## 模块概述

### 核心功能
智能推荐模块基于霍兰德职业兴趣理论，结合学生的学业表现、技能水平和市场需求，提供三大核心推荐服务：

```
智能推荐模块
├── 实习岗位匹配推荐
├── 技能培训课程推荐
└── 职业发展路径规划
```

### 算法1：实习岗位匹配算法

#### 1.1 混合推荐策略

```python
# 综合匹配度计算公式
Match_Score = α × Holland_Score + β × Skill_Score + γ × Academic_Score + δ × Preference_Score

# 权重分配
α = 0.35  # 霍兰德匹配度权重
β = 0.30  # 技能匹配度权重
γ = 0.20  # 学业匹配度权重
δ = 0.15  # 偏好匹配度权重
```

#### 1.2 霍兰德类型匹配算法

**方法一：余弦相似度**

```python
import numpy as np
from sklearn.metrics.pairwise import cosine_similarity

def holland_match_score(student_holland, job_holland):
    """
    计算学生与岗位的霍兰德类型匹配度
    
    Args:
        student_holland: dict, 学生的六维霍兰德得分 {'R': 45, 'I': 85, ...}
        job_holland: dict, 岗位的霍兰德权重 {'I': 0.7, 'C': 0.2, 'E': 0.1}
    
    Returns:
        float: 匹配度得分 (0-100)
    """
    # 将字典转为向量（按RIASEC顺序）
    holland_types = ['R', 'I', 'A', 'S', 'E', 'C']
    
    student_vector = np.array([student_holland.get(t, 0) for t in holland_types])
    
    # 岗位向量：将权重转换为100分制
    job_vector = np.zeros(6)
    for t, weight in job_holland.items():
        idx = holland_types.index(t)
        job_vector[idx] = weight * 100
    
    # 计算余弦相似度
    similarity = cosine_similarity(
        student_vector.reshape(1, -1),
        job_vector.reshape(1, -1)
    )[0][0]
    
    # 转换为0-100分
    return similarity * 100

# 示例
student = {'R': 45, 'I': 85, 'A': 60, 'S': 50, 'E': 70, 'C': 75}
job = {'I': 0.7, 'C': 0.2, 'E': 0.1}
score = holland_match_score(student, job)
print(f"霍兰德匹配度: {score:.2f}分")  # 输出: 88.25分
```

**方法二：加权欧氏距离（考虑岗位权重）**

```python
def holland_weighted_distance(student_holland, job_holland):
    """
    使用加权欧氏距离计算匹配度
    距离越小，匹配度越高
    """
    holland_types = ['R', 'I', 'A', 'S', 'E', 'C']
    
    distance = 0
    for t in holland_types:
        student_score = student_holland.get(t, 0)
        job_weight = job_holland.get(t, 0)
        job_score = job_weight * 100
        
        # 加权距离
        distance += job_weight * (student_score - job_score) ** 2
    
    distance = np.sqrt(distance)
    
    # 转换为匹配度得分（距离越小，得分越高）
    max_distance = 100  # 理论最大距离
    match_score = (1 - distance / max_distance) * 100
    
    return max(0, match_score)
```

#### 1.3 技能匹配算法

```python
def skill_match_score(student_skills, job_requirements):
    """
    计算技能匹配度
    
    Args:
        student_skills: dict, 学生技能及等级
        job_requirements: dict, 岗位技能要求
            {
                '必需技能': {'Python': 3, 'SQL': 3},
                '加分技能': {'机器学习': 2}
            }
    
    Returns:
        float: 技能匹配度得分 (0-100)
    """
    # 必需技能匹配
    required = job_requirements.get('必需技能', {})
    required_score = 0
    required_weight = 0.7
    
    for skill, required_level in required.items():
        student_level = student_skills.get(skill, {}).get('level', 0)
        
        if student_level >= required_level:
            # 满足要求得满分，超过要求额外加分
            required_score += 1 + 0.1 * (student_level - required_level)
        else:
            # 未达到要求，按比例计分
            required_score += student_level / required_level * 0.6
    
    if len(required) > 0:
        required_score = (required_score / len(required)) * 100
    else:
        required_score = 100
    
    # 加分技能匹配
    bonus = job_requirements.get('加分技能', {})
    bonus_score = 0
    bonus_weight = 0.3
    
    for skill, required_level in bonus.items():
        student_level = student_skills.get(skill, {}).get('level', 0)
        
        if student_level >= required_level:
            bonus_score += 1
    
    if len(bonus) > 0:
        bonus_score = (bonus_score / len(bonus)) * 100
    else:
        bonus_score = 0
    
    # 综合得分
    total_score = required_score * required_weight + bonus_score * bonus_weight
    
    return min(100, total_score)

# 示例
student_skills = {
    'Python': {'level': 4, 'certified': True},
    'SQL': {'level': 3, 'certified': False},
    '机器学习': {'level': 3, 'certified': False}
}
job_req = {
    '必需技能': {'Python': 3, 'SQL': 3},
    '加分技能': {'机器学习': 2, '大数据处理': 2}
}

print(f"技能匹配度: {score:.2f}分")
```

#### 1.4 学业匹配算法

```python
def academic_match_score(student_profile, job_requirements):
    """
    计算学业表现匹配度
    """
    score = 0
    
    # GPA匹配（40%权重）
    student_gpa = student_profile['basic_info']['gpa']
    min_gpa = job_requirements['min_gpa']
    
    if student_gpa >= min_gpa:
        gpa_score = min(100, (student_gpa / 4.0) * 100)
    else:
        gpa_score = (student_gpa / min_gpa) * 60  # 未达标打折
    
    score += gpa_score * 0.4
    
    # 年级匹配（30%权重）
    student_grade = student_profile['basic_info']['grade']
    grade_range = job_requirements['grade_range']
    
    if grade_range[0] <= student_grade <= grade_range[1]:
        grade_score = 100
    else:
        grade_score = 50  # 年级不匹配减分
    
    score += grade_score * 0.3
    
    # 专业匹配（30%权重）
    student_major = student_profile['basic_info']['major']
    preferred_majors = job_requirements['preferred_majors']
    
    # 使用关键词匹配
    major_score = 0
    for preferred in preferred_majors:
        if preferred in student_major or student_major in preferred:
            major_score = 100
            break
    else:
        major_score = 60  # 专业不完全匹配
    
    score += major_score * 0.3
    
    return score
```

#### 1.5 偏好匹配算法

```python
def preference_match_score(student_preferences, job_details):
    """
    计算学生偏好与岗位的匹配度
    """
    score = 0
    
    # 地点偏好（35%权重）
    if job_details['location'] in student_preferences['work_location']:
        location_score = 100
    else:
        location_score = 40
    score += location_score * 0.35
    
    # 公司规模偏好（25%权重）
    if job_details['company_size'] in student_preferences['company_size']:
        size_score = 100
    else:
        size_score = 60
    score += size_score * 0.25
    
    # 行业偏好（25%权重）
    # 需要从岗位所属行业判断
    industry_score = 70  # 默认中等匹配
    score += industry_score * 0.25
    
    # 工作生活平衡（15%权重）
    wlb_preference = student_preferences['work_life_balance']
    # 假设根据公司和岗位评估WLB得分
    job_wlb = 7  # 从历史数据或公司评价获取
    wlb_score = max(0, 100 - abs(wlb_preference - job_wlb) * 10)
    score += wlb_score * 0.15
    
    return score
```

#### 1.6 完整推荐系统

```python
class InternshipRecommendationSystem:
    """实习岗位推荐系统"""
    
    def __init__(self, alpha=0.35, beta=0.30, gamma=0.20, delta=0.15):
        self.alpha = alpha  # 霍兰德权重
        self.beta = beta    # 技能权重
        self.gamma = gamma  # 学业权重
        self.delta = delta  # 偏好权重
    
    def calculate_match_score(self, student, job):
        """计算综合匹配度"""
        
        # 1. 霍兰德匹配度
        holland_score = holland_match_score(
            student['holland_scores'],
            job['holland_match']
        )
        
        # 2. 技能匹配度
        skill_score = skill_match_score(
            student['skills']['programming'],
            job['required_skills']
        )
        
        # 3. 学业匹配度
        academic_score = academic_match_score(
            student,
            job['requirements']
        )
        
        # 4. 偏好匹配度
        preference_score = preference_match_score(
            student['career_preferences'],
            job['job_details']
        )
        
        # 综合得分
        total_score = (
            self.alpha * holland_score +
            self.beta * skill_score +
            self.gamma * academic_score +
            self.delta * preference_score
        )
        
        return {
            'total_score': total_score,
            'holland_score': holland_score,
            'skill_score': skill_score,
            'academic_score': academic_score,
            'preference_score': preference_score
        }
    
    def recommend_internships(self, student, job_list, top_n=10):
        """
        为学生推荐实习岗位
        
        Returns:
            list: 推荐岗位列表，按匹配度降序排列
        """
        recommendations = []
        
        for job in job_list:
            scores = self.calculate_match_score(student, job)
            
            recommendations.append({
                'job_id': job['job_id'],
                'company': job['company'],
                'position': job['position'],
                'match_score': scores['total_score'],
                'score_breakdown': scores,
                'reasons': self._generate_match_reasons(scores, student, job)
            })
        
        # 按匹配度排序
        recommendations.sort(key=lambda x: x['match_score'], reverse=True)
        
        return recommendations[:top_n]
    
    def _generate_match_reasons(self, scores, student, job):
        """生成推荐理由"""
        reasons = []
        
        # 根据各项得分生成推荐理由
        if scores['holland_score'] >= 80:
            holland_code = student['holland_code']
            reasons.append(f"您的职业兴趣类型({holland_code})与该岗位高度匹配")
        
        if scores['skill_score'] >= 80:
            reasons.append("您的技能完全满足岗位要求")
        elif scores['skill_score'] >= 60:
            reasons.append("您的核心技能基本满足要求")
        
        if scores['academic_score'] >= 85:
            reasons.append("您的学业表现优秀")
        
        if scores['preference_score'] >= 75:
            reasons.append("岗位地点和公司类型符合您的偏好")
        
        return reasons

# 使用示例
recommender = InternshipRecommendationSystem()
recommendations = recommender.recommend_internships(student_profile, job_list, top_n=10)

for i, rec in enumerate(recommendations, 1):
    print(f"\n{i}. {rec['company']} - {rec['position']}")
    print(f"   匹配度: {rec['match_score']:.2f}分")
    print(f"   推荐理由: {', '.join(rec['reasons'])}")
```

### 算法2：课程推荐算法

#### 2.1 技能缺口分析

```python
def analyze_skill_gap(student, target_jobs):
    """
    分析学生当前技能与目标岗位的差距
    
    Args:
        student: 学生画像
        target_jobs: 学生感兴趣的岗位列表
    
    Returns:
        dict: 技能缺口分析结果
    """
    # 收集目标岗位所需技能
    required_skills = {}
    
    for job in target_jobs:
        for skill_type, skills in job['required_skills'].items():
            for skill, level in skills.items():
                if skill not in required_skills:
                    required_skills[skill] = {'max_level': level, 'frequency': 1}
                else:
                    required_skills[skill]['max_level'] = max(
                        required_skills[skill]['max_level'], level
                    )
                    required_skills[skill]['frequency'] += 1
    
    # 分析技能差距
    skill_gaps = []
    
    all_student_skills = {}
    for category in student['skills'].values():
        if isinstance(category, dict):
            all_student_skills.update(category)
    
    for skill, info in required_skills.items():
        current_level = 0
        if skill in all_student_skills:
            if isinstance(all_student_skills[skill], dict):
                current_level = all_student_skills[skill].get('level', 0)
            else:
                current_level = all_student_skills[skill]
        
        gap = info['max_level'] - current_level
        
        if gap > 0:
            # 计算优先级（差距越大、出现频率越高，优先级越高）
            priority = gap * info['frequency']
            
            skill_gaps.append({
                'skill': skill,
                'current_level': current_level,
                'target_level': info['max_level'],
                'gap': gap,
                'frequency': info['frequency'],
                'priority': priority
            })
    
    # 按优先级排序
    skill_gaps.sort(key=lambda x: x['priority'], reverse=True)
    
    return skill_gaps

#     {'skill': '大数据处理', 'current_level': 0, 'target_level': 3, 'gap': 3, 'frequency': 3, 'priority': 9},
# ]
```

```python
class CourseRecommendationSystem:
    """课程推荐系统"""
    
    def recommend_courses(self, student, skill_gaps, course_catalog, top_n=5):
        """
        基于技能缺口推荐课程
        
        Args:
            student: 学生画像
            skill_gaps: 技能缺口列表
            course_catalog: 课程目录
            top_n: 推荐课程数量
        
        Returns:
            list: 推荐课程列表
        """
        recommendations = []
        
        for course in course_catalog:
            score = self._calculate_course_score(student, course, skill_gaps)
            
            if score > 0:
                recommendations.append({
                    'course_id': course['course_id'],
                    'title': course['title'],
                    'platform': course['platform'],
                    'score': score,
                    'reasons': self._generate_course_reasons(course, skill_gaps),
                    'learning_path': self._generate_learning_path(course, student)
                })
        
        # 排序并返回
        recommendations.sort(key=lambda x: x['score'], reverse=True)
        return recommendations[:top_n]
    
    def _calculate_course_score(self, student, course, skill_gaps):
        """计算课程推荐分数"""
        score = 0
        
        # 1. 技能相关性（50%权重）
        skill_relevance = 0
        for skill, improvement in course['skill_improvement'].items():
            for gap in skill_gaps:
                if gap['skill'] == skill:
                    # 优先级越高，得分越高
                    skill_relevance += gap['priority'] * improvement * 10
                    break
        score += skill_relevance * 0.5
        
        # 2. 霍兰德匹配度（20%权重）
        holland_match = 0
        student_top_codes = student['holland_code']
        for code in student_top_codes:
            if code in course['holland_fit']:
                holland_match += 1
        holland_match = (holland_match / len(student_top_codes)) * 100
        score += holland_match * 0.2
        
        # 3. 前置条件满足度（15%权重）
        prerequisite_met = self._check_prerequisites(student, course)
        score += prerequisite_met * 0.15
        
        # 4. 课程质量（15%权重）
        quality_score = (
            course['rating'] / 5.0 * 50 +  # 评分
            course['completion_rate'] * 50   # 完成率
        )
        score += quality_score * 0.15
        
        return score
    
    def _check_prerequisites(self, student, course):
        """检查前置条件"""
        if not course.get('prerequisites'):
            return 100
        
        met_count = 0
        total_count = len(course['prerequisites'])
        
        all_student_skills = {}
        for category in student['skills'].values():
            if isinstance(category, dict):
                all_student_skills.update(category)
        
        for skill, required_level in course['prerequisites'].items():
            current_level = 0
            if skill in all_student_skills:
                if isinstance(all_student_skills[skill], dict):
                    current_level = all_student_skills[skill].get('level', 0)
                else:
                    current_level = all_student_skills[skill]
            
            if current_level >= required_level:
                met_count += 1
        
        return (met_count / total_count) * 100
    
    def _generate_course_reasons(self, course, skill_gaps):
        """生成推荐理由"""
        reasons = []
        
        # 找出该课程可以填补的技能缺口
        for skill in course['skill_improvement'].keys():
            for gap in skill_gaps[:3]:  # 只看前3个优先级最高的
                if gap['skill'] == skill:
                    reasons.append(f"提升您的{skill}技能（当前Lv{gap['current_level']} → 目标Lv{gap['target_level']}）")
                    break
        
        if course['rating'] >= 4.5:
            reasons.append(f"高质量课程（评分{course['rating']}⭐）")
        
        if course['completion_rate'] >= 0.75:
            reasons.append(f"完成率高（{course['completion_rate']*100:.0f}%学员完成学习）")
        
        return reasons
    
    def _generate_learning_path(self, course, student):
        """生成学习路径建议"""
        return {
            'estimated_duration': f"{course['duration']}小时",
            'weekly_commitment': "建议每周学习5-8小时",
            'completion_date': "预计4-6周完成",
            'next_steps': "完成后建议进行项目实践"
        }
```

```python
class CareerPathPlanner:
    """职业发展路径规划系统"""
    
    def __init__(self, career_paths_db):
        self.career_paths_db = career_paths_db
    
    def recommend_career_paths(self, student, top_n=3):
        """
        为学生推荐职业发展路径
        
        Returns:
            list: 推荐的职业路径及详细规划
        """
        path_scores = []
        
        for path in self.career_paths_db:
            score = self._calculate_path_fit(student, path)
            
            path_scores.append({
                'path_id': path['path_id'],
                'path_name': path['path_name'],
                'fit_score': score,
                'path_details': path,
                'personalized_plan': self._generate_personalized_plan(student, path)
            })
        
        # 按适合度排序
        path_scores.sort(key=lambda x: x['fit_score'], reverse=True)
        
        return path_scores[:top_n]
    
    def _calculate_path_fit(self, student, path):
        """计算路径适配度"""
        score = 0
        
        # 1. 霍兰德类型匹配（40%权重）
        student_codes = set(student['holland_code'])
        path_codes = set(path['holland_codes'])
        holland_overlap = len(student_codes & path_codes) / len(path_codes)
        score += holland_overlap * 100 * 0.4
        
        # 2. 当前技能基础（30%权重）
        # 评估学生当前技能与路径初期要求的匹配度
        first_stage = path['stages'][0]
        current_skill_fit = self._assess_current_readiness(student, first_stage)
        score += current_skill_fit * 0.3
        
        # 3. 职业偏好匹配（20%权重）
        career_options = path['stages'][-1].get('career_options', [])
        preference_match = 0
        # 这里可以与学生的职业偏好进行匹配
        # 简化处理：假设有一定匹配度
        preference_match = 70  # 默认值
        score += preference_match * 0.2
        
        # 4. 成功案例相似度（10%权重）
        if 'success_cases' in path and len(path['success_cases']) > 0:
            similarity = self._calculate_case_similarity(student, path['success_cases'])
            score += similarity * 0.1
        
        return score
    
    def _assess_current_readiness(self, student, first_stage):
        """评估学生当前对路径的准备度"""
        milestones = first_stage.get('milestones', {})
        
        if not milestones:
            return 70  # 默认中等准备度
        
        readiness = 0
        total_criteria = 0
        
        # 检查技能达标情况
        if '技能达标' in milestones:
            total_criteria += 1
            skill_requirements = milestones['技能达标']
            
            all_student_skills = {}
            for category in student['skills'].values():
                if isinstance(category, dict):
                    all_student_skills.update(category)
            
            met = 0
            for skill_req in skill_requirements:
                # 解析形如 "Python>=4"
                skill_name = skill_req.split('>=')[0]
                required_level = int(skill_req.split('>=')[1])
                
                current_level = 0
                if skill_name in all_student_skills:
                    if isinstance(all_student_skills[skill_name], dict):
                        current_level = all_student_skills[skill_name].get('level', 0)
                    else:
                        current_level = all_student_skills[skill_name]
                
                if current_level >= required_level:
                    met += 1
            
            if len(skill_requirements) > 0:
                readiness += (met / len(skill_requirements)) * 100
        
        # 检查项目经验
        if '项目' in milestones:
            total_criteria += 1
            required_projects = milestones['项目']
            student_projects = len(student['academic_records']['projects'])
            
            if student_projects >= required_projects:
                readiness += 100
            else:
                readiness += (student_projects / required_projects) * 60
        
        if total_criteria > 0:
            return readiness / total_criteria
        return 70
    
    def _calculate_case_similarity(self, student, success_cases):
        """计算与成功案例的相似度"""
        # 简化：基于GPA和学校等级的相似度
        avg_satisfaction = sum(case['career_satisfaction'] for case in success_cases) / len(success_cases)
        
        # 满意度越高，该路径越可靠
        return avg_satisfaction * 10  # 转换为百分制
    
    def _generate_personalized_plan(self, student, path):
        """
        生成个性化的职业发展计划
        """
        current_grade = student['basic_info']['grade']
        
        personalized_stages = []
        
        for stage in path['stages']:
            # 根据学生当前状态调整时间线
            adjusted_stage = stage.copy()
            
            # 分析当前阶段的完成度
            completion = self._assess_stage_completion(student, stage)
            
            # 生成具体行动计划
            action_plan = self._generate_action_plan(student, stage, completion)
            
            adjusted_stage['current_completion'] = completion
            adjusted_stage['action_plan'] = action_plan
            
            personalized_stages.append(adjusted_stage)
        
        return {
            'stages': personalized_stages,
            'overall_readiness': self._assess_current_readiness(student, path['stages'][0]),
            'estimated_success_rate': self._estimate_success_rate(student, path)
        }
    
    def _assess_stage_completion(self, student, stage):
        """评估阶段完成度"""
        milestones = stage.get('milestones', {})
        
        if not milestones:
            return 0
        
        completion = 0
        total = 0
        
        # 评估各个里程碑
        for milestone_type, requirements in milestones.items():
            total += 1
            
            if milestone_type == '技能达标':
                skill_met = 0
                for req in requirements:
                    # 检查是否满足
                    # (简化处理)
                    skill_met += 0.5  # 部分满足
                completion += (skill_met / len(requirements))
            
            elif milestone_type == '证书':
                # 检查学生已获得的证书
                completion += 0  # 假设未获得
            
            elif milestone_type == '项目':
                student_projects = len(student['academic_records']['projects'])
                required = requirements
                if student_projects >= required:
                    completion += 1
                else:
                    completion += student_projects / required
        
        if total > 0:
            return (completion / total) * 100
        return 0
    
    def _generate_action_plan(self, student, stage, current_completion):
        """生成具体行动计划"""
        actions = []
        
        # 基于完成度生成不同的建议
        if current_completion < 30:
            actions.append({
                'priority': '高',
                'action': f"立即开始学习{stage['courses'][0]}的相关课程",
                'deadline': '1个月内'
            })
            actions.append({
                'priority': '高',
                'action': f"寻找{stage['recommended_internships'][0]}相关的实习机会",
                'deadline': '2个月内'
            })
        elif current_completion < 70:
            actions.append({
                'priority': '中',
                'action': "继续提升专业技能，准备相关认证考试",
                'deadline': '3个月内'
            })
            actions.append({
                'priority': '中',
                'action': "参与1-2个实战项目，积累经验",
                'deadline': '当前学期'
            })
        else:
            actions.append({
                'priority': '低',
                'action': "保持技能更新，关注行业动态",
                'deadline': '持续进行'
            })
            actions.append({
                'priority': '高',
                'action': f"准备进入下一阶段：{stage.get('stage', '未知阶段')}",
                'deadline': '下学期'
            })
        
        return actions
    
    def _estimate_success_rate(self, student, path):
        """
        估算成功率
        基于历史数据和当前学生情况
        """
        # 简化：基于GPA、技能匹配度等因素
        base_rate = 60  # 基础成功率
        
        # GPA加成
        if student['basic_info']['gpa'] >= 3.5:
            base_rate += 15
        elif student['basic_info']['gpa'] >= 3.0:
            base_rate += 10
        
        # 技能准备度加成
        readiness = self._assess_current_readiness(student, path['stages'][0])
        base_rate += (readiness - 70) * 0.3  # 超过70分每分加0.3
        
        return min(95, max(40, base_rate))  # 限制在40-95之间
```

### 后端API设计

```python
from flask import Flask, jsonify, request
from flask_cors import CORS

app = Flask(__name__)
CORS(app)

# 初始化推荐系统
internship_recommender = InternshipRecommendationSystem()
course_recommender = CourseRecommendationSystem()
career_planner = CareerPathPlanner(career_paths_database)

@app.route('/api/recommendations/internships', methods=['POST'])
def get_internship_recommendations():
    """
    获取实习岗位推荐
    
    Request Body:
    {
        "student_id": "2024001",
        "filters": {
            "location": ["北京", "上海"],
            "industry": ["互联网"],
            "min_match_score": 80
        },
        "top_n": 10
    }
    
    Response:
    {
        "success": true,
        "data": [
            {
                "job_id": "J2024001",
                "company": "字节跳动",
                "position": "数据分析实习生",
                "match_score": 92.5,
                "score_breakdown": {...},
                "reasons": [...]
            }
        ]
    }
    """
    data = request.json
    student_id = data.get('student_id')
    filters = data.get('filters', {})
    top_n = data.get('top_n', 10)
    
    # 获取学生画像
    student = get_student_profile(student_id)
    
    # 获取岗位列表（应用筛选条件）
    jobs = get_filtered_jobs(filters)
    
    # 生成推荐
    recommendations = internship_recommender.recommend_internships(
        student, jobs, top_n
    )
    
    return jsonify({
        'success': True,
        'data': recommendations
    })

@app.route('/api/recommendations/courses', methods=['POST'])
def get_course_recommendations():
    """
    获取课程推荐
    
    Request Body:
    {
        "student_id": "2024001",
        "target_jobs": ["J2024001", "J2024002"],
        "top_n": 5
    }
    """
    data = request.json
    student_id = data.get('student_id')
    target_job_ids = data.get('target_jobs', [])
    top_n = data.get('top_n', 5)
    
    student = get_student_profile(student_id)
    target_jobs = get_jobs_by_ids(target_job_ids)
    
    # 分析技能缺口
    skill_gaps = analyze_skill_gap(student, target_jobs)
    
    # 推荐课程
    course_catalog = get_course_catalog()
    recommendations = course_recommender.recommend_courses(
        student, skill_gaps, course_catalog, top_n
    )
    
    return jsonify({
        'success': True,
        'data': {
            'skill_gaps': skill_gaps,
            'course_recommendations': recommendations
        }
    })

@app.route('/api/recommendations/career-paths', methods='POST'])
def get_career_path_recommendations():
    """
    获取职业发展路径推荐
    """
    data = request.json
    student_id = data.get('student_id')
    top_n = data.get('top_n', 3)
    
    student = get_student_profile(student_id)
    recommendations = career_planner.recommend_career_paths(student, top_n)
    
    return jsonify({
        'success': True,
        'data': recommendations
    })

if __name__ == '__main__':
    app.run(debug=True, port=5000)
```

## 总结

这个智能推荐模块设计包含：

### 核心算法
1. **实习岗位推荐** - 基于霍兰德类型、技能、学业和偏好的混合推荐
2. **课程推荐** - 基于技能缺口分析的个性化学习路径
3. **职业路径规划** - 长期职业发展的分阶段规划

### 技术特点
- 多维度匹配算法（霍兰德+技能+学业+偏好）
- 动态技能缺口分析
- 个性化行动计划生成
- 可视化展示（雷达图、时间线、进度条）

### 创新点
- 将霍兰德理论与数据科学方法结合
- 提供可操作的具体行动建议
- 基于历史成功案例的路径优化
- 实时跟踪学生成长轨迹

该系统可以有效帮助大学生从入学开始进行职业规划，在实习阶段实现精准分流！
